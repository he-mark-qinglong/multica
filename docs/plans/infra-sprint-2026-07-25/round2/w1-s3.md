```python
"""Performance budget + large-input sanity for _shared.run_backtest.

Regression guard only — the 60s budget is ~10x the expected runtime of the
vectorised engine on 2.4M bars; it exists to catch accidental de-vectorisation,
not to enforce tuning targets.
"""
from __future__ import annotations

import sys
import time
from pathlib import Path

import numpy as np
import pandas as pd

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))  # quant-loop root

from _shared.run_backtest import Trade, run_backtest  # noqa: E402

PERF_BUDGET_SECONDS = 60.0


def _make_bars(n_bars: int, seed: int = 7) -> pd.DataFrame:
    idx = pd.date_range("2021-01-01", periods=n_bars, freq="1min", tz="UTC")
    rng = np.random.default_rng(seed)
    ret = 1e-5 + 2e-3 * rng.standard_normal(n_bars)
    close = 100.0 * np.cumprod(1.0 + ret)
    return pd.DataFrame({"close": close}, index=idx)


def _make_trades(idx: pd.DatetimeIndex, step: int = 240, hold: int = 60) -> list[Trade]:
    trades: list[Trade] = []
    i = 0
    k = 0
    while i + hold < len(idx):
        trades.append(Trade(
            entry_ts=idx[i], exit_ts=idx[i + hold],
            direction="long" if k % 2 == 0 else "short",
        ))
        i += step
        k += 1
    return trades


def test_perf_2p4m_bars_10k_trades_under_budget():
    bars = _make_bars(2_400_000)
    trades = _make_trades(bars.index)          # ~10_000 trades
    assert len(trades) >= 9_000
    t0 = time.perf_counter()
    result = run_backtest(bars, trades, cost_bps_rt=4.0,
                          cost_mode="fill", freq_per_year=365 * 24 * 60)
    elapsed = time.perf_counter() - t0
    assert result["n_trades"] >= 9_000
    assert len(result["equity"]) == len(bars)
    assert elapsed < PERF_BUDGET_SECONDS, f"run_backtest took {elapsed:.1f}s"


def test_100k_bars_result_structure_sanity():
    bars = _make_bars(100_000)
    trades = _make_trades(bars.index)          # ~416 trades
    result = run_backtest(bars, trades, cost_bps_rt=4.0,
                          cost_mode="fill", freq_per_year=365 * 24 * 60)
    assert result["n_skipped"] == 0            # all trades are on-bar by construction
    assert result["n_trades"] == len(trades)
    eq = result["equity"]
    assert isinstance(eq, pd.Series) and eq.index.equals(bars.index)
    assert {"sharpe", "annualised_pct", "total_return_pct", "max_drawdown_pct"} \
        <= set(result["metrics"])
    assert np.isfinite(eq.to_numpy()).all()
```

注意：交易全在 bar 上（entry/exit 直接取 index 元素），所以 `n_skipped` 必须为 0；若引擎改了 off-bar 语义导致这里红，说明引擎变了，不是你测试错了——如实报告。
